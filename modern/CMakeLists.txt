cmake_minimum_required(VERSION 3.20)
project(SourceNavigator VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Test Core5Compat PrintSupport)
set(QT_VERSION_MAJOR 6)
add_executable(icon-builder EXCLUDE_FROM_ALL tools/generate_icon.cpp)
target_link_libraries(icon-builder PRIVATE Qt6::Gui)

set(SCINTILLA "${CMAKE_CURRENT_SOURCE_DIR}/third_party/scintilla")
set(LEXILLA "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lexilla")
file(GLOB SCI_SOURCES CONFIGURE_DEPENDS "${SCINTILLA}/src/*.cxx")
add_library(scintilla STATIC ${SCI_SOURCES}
    "${SCINTILLA}/qt/ScintillaEditBase/PlatQt.cpp"
    "${SCINTILLA}/qt/ScintillaEditBase/ScintillaQt.cpp"
    "${SCINTILLA}/qt/ScintillaEditBase/ScintillaEditBase.cpp"
    "${SCINTILLA}/qt/ScintillaEditBase/ScintillaEditBase.h"
    "${SCINTILLA}/qt/ScintillaEditBase/ScintillaQt.h")
target_include_directories(scintilla PUBLIC "${SCINTILLA}/include" "${SCINTILLA}/src" "${SCINTILLA}/qt/ScintillaEditBase")
target_compile_definitions(scintilla PUBLIC EXPORT_IMPORT_API= PRIVATE SCINTILLA_QT=1)
target_link_libraries(scintilla PUBLIC Qt${QT_VERSION_MAJOR}::Widgets)
target_link_libraries(scintilla PUBLIC Qt6::Core5Compat)
file(GLOB LEX_SOURCES CONFIGURE_DEPENDS "${LEXILLA}/lexlib/*.cxx" "${LEXILLA}/lexers/*.cxx")
add_library(lexilla STATIC ${LEX_SOURCES} "${LEXILLA}/src/Lexilla.cxx")
set_target_properties(lexilla PROPERTIES AUTOMOC OFF)
target_include_directories(lexilla PUBLIC "${LEXILLA}/include" "${SCINTILLA}/include" PRIVATE "${LEXILLA}/lexlib")
add_library(navigator-ui STATIC gui/theme.cpp gui/editor.cpp gui/window.cpp gui/window_actions.cpp gui/window_features.cpp gui/printing.cpp gui/localization.cpp gui/icons.qrc)
target_link_libraries(navigator-ui PUBLIC Qt6::PrintSupport scintilla lexilla Qt${QT_VERSION_MAJOR}::Widgets)
target_include_directories(navigator-ui PUBLIC gui)
target_compile_definitions(navigator-ui PRIVATE QT_NO_CAST_FROM_ASCII)
target_compile_definitions(navigator-ui PUBLIC SOURCE_NAVIGATOR_VERSION=\"${PROJECT_VERSION}\")
add_executable(source-navigator WIN32 gui/main.cpp gui/crash.cpp)
if(WIN32)
    enable_language(RC)
    target_sources(source-navigator PRIVATE assets/application.rc)
    target_include_directories(source-navigator PRIVATE assets)
endif()
target_link_libraries(source-navigator PRIVATE navigator-ui)
if(MINGW)
    target_link_options(source-navigator PRIVATE -Wl,--dynamicbase,--nxcompat -static-libgcc -static-libstdc++)
endif()
enable_testing()
add_executable(external-recorder tests/external_recorder.cpp)
target_link_libraries(external-recorder PRIVATE Qt6::Core)
add_executable(ui-tests tests/ui_test.cpp)
target_link_libraries(ui-tests PRIVATE navigator-ui Qt${QT_VERSION_MAJOR}::Test)
add_test(NAME ui-tests COMMAND ui-tests -platform offscreen -o qt-results.txt,txt)
install(TARGETS source-navigator RUNTIME DESTINATION .)
